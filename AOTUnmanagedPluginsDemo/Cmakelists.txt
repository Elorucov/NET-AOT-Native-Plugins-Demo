cmake_minimum_required(VERSION 3.20)
project(plugin C)

# Builds shared library / DLL
add_library(plugin SHARED plugin.c)

# Exports only functions that marked as EXPORT
if(NOT MSVC)
    target_compile_options(plugin PRIVATE -fvisibility=hidden)
endif()

set_target_properties(plugin PROPERTIES
    OUTPUT_NAME "plugin"          # → libplugin.so / libplugin.dylib / plugin.dll
    C_STANDARD  11
)
